    const ModalUI = {

                if(!s) return;

                try {
                    // Send via GET request
                    await fetch(`/api/wifi?ssid=${encodeURIComponent(s)}&pass=${encodeURIComponent(p)}`);
                    stat.textContent = "Device connecting... Do not leave this page.";
                    
                    // START POLLING FOR NEW IP ADDRESS
                    const checkInterval = setInterval(async () => {
                        try {
                            const res = await fetch('/api/data');
                            const json = await res.json();
                            
                            // IF WE GET A VALID NEW IP
                            if (json.ip && json.ip !== "0.0.0.0" && json.ip !== "192.168.4.1") {
                                stat.textContent = "Connected! Redirecting to " + json.ip + "...";
                                clearInterval(checkInterval);
                                // FORCE REDIRECT TO NEW ADDRESS
                                setTimeout(() => {
                                    window.location.href = "http://" + json.ip;
                                }, 1500);
                            }
                        } catch(e) {
                            // Ignore errors while waiting
                        }
                    }, 2000); // Check every 2 seconds

                } catch(e) {
                    stat.textContent = "Error sending credentials.";
                    btn.disabled = false;
                }
            };
        }
    };

    // ==================== API.JS ====================
    const FiaphyAPI = {
        config: { apiEndpoint: '/api/data', updateInterval: 2000 },
        isConnected: false,
        init() { this.checkConnection(); },
        async checkConnection() {
            try { this.isConnected = true; return true; } catch (e) { this.isConnected = false; return false; }
        },
        async fetchData() {
            // Added Timeout to prevent hanging
            const controller = new AbortController();
            const id = setTimeout(() => controller.abort(), 3000); // 3s Timeout
            try {
                const response = await fetch(this.config.apiEndpoint, { signal: controller.signal });
                clearTimeout(id);
                if (!response.ok) throw new Error("Network response was not ok");
                const data = await response.json();
                this.isConnected = true;
                return data;
            } catch (error) {
                this.isConnected = false;
                throw error;
            }
        },
        startPolling(callback) {
            const poll = async () => {
                try {
                    const data = await this.fetchData();
                    if (callback) callback(data);
                } catch (error) {
                    if (callback) callback(null, error);
                }
            };
            poll();
            return setInterval(poll, this.config.updateInterval);
        },
        stopPolling(id) { if (id) clearInterval(id); }
    };

        updateConnectionStatus(isConnected) {

            const text = this.elements.statusText;
            if (!status) return;
            if (isConnected) { status.classList.add('connected'); status.classList.remove('disconnected'); text.textContent = 'Connected'; }
            else { status.classList.remove('connected'); status.classList.add('disconnected'); text.textContent = 'Disconnected'; }

        toggleNoDevice(show) {
            if (show) { this.elements.noDevice?.classList.remove('hidden'); this.elements.dataGrid?.classList.add('hidden'); }

        },
        updateData(data) {
            if (!data) { this.toggleNoDevice(true); this.updateConnectionStatus(false); return; }
            this.toggleNoDevice(false);
            this.updateConnectionStatus(true);
            this.updateValue('temperature', data.temperature, 1);
            this.updateValue('humidity', data.humidity, 0);
            this.updateValue('airPressure', data.airPressure, 1);
            this.updateValue('solarRadiation', data.solarRadiation, 0);
            this.updateValue('heatFlux', data.heatFlux, 1);
            this.updateValue('deltaTemperature', data.deltaTemperature, 2);
        },

float apply_filter(float new_val, float old_val) {
    if (old_val == 0.0f) return new_val;
    return (old_val * (1.0f - FILTER_FACTOR)) + (new_val * FILTER_FACTOR);
}

String getUrlParam(String request, String param) {
    String p = param + "=";
    int start = request.indexOf(p);
    if (start == -1) return "";
    start += p.length();
    int end = request.indexOf("&", start);
    if (end == -1) end = request.indexOf(" ", start);
    String value = request.substring(start, end);
    value.replace("%20", " ");
    return value;
}

void softStartSystem(float t, float h, float p) {
    Serial.println("Performing Soft-Start Ramp (200 Steps)...");
    for (int i = 1; i <= 200; i++) { 
        float factor = (float)i / 200.0f;
        dtdss.feedReferenceTemperature(t * factor, 0);
        dtdss.feedReferenceHumidity(h * factor, 0);
        dtdss.feedReferencePressure(p * factor, 0);
        dtdss.feedFluxTemperature(t * factor, 0);
        dtdss.feedFluxHumidity(h * factor, 0);
        dtdss.feedFluxPressure(p * factor, 0);
        delay(10); 
    }
    Serial.println("Soft-Start Complete.");
}

void sendAT(String command, int waitMs) {
    Serial1.println(command);
    unsigned long start = millis();
    while (millis() - start < waitMs) {
        while (Serial1.available()) Serial.write(Serial1.read()); 
    }
}

// Helper: Block processing until OK or ERROR
bool sendATWaitForResponse(String command, int timeout) {
    Serial1.println(command);
    unsigned long start = millis();
    while(millis() - start < timeout) {
        if(Serial1.available()) {
            String resp = Serial1.readString();
            Serial.print(resp); // Echo for debugging
            if(resp.indexOf("OK") != -1) return true;
            if(resp.indexOf("ERROR") != -1) return false;
            if(resp.indexOf("FAIL") != -1) return false;
        }
    }
    return false;
}

bool sendChunked(int connectionId, const char* data, bool isProgMem) {
    int len = isProgMem ? strlen_P(data) : strlen(data);
    int chunkSize = 512; 
    
    for (int i = 0; i < len; i += chunkSize) {
        int end = (i + chunkSize < len) ? i + chunkSize : len;
        int chunkLen = end - i;
        
        char buf[chunkSize + 1];
        if (isProgMem) {
            strncpy_P(buf, data + i, chunkLen);
        } else {
            strncpy(buf, data + i, chunkLen);
        }
        buf[chunkLen] = '\0';

        Serial1.print("AT+CIPSEND=");
        Serial1.print(connectionId);
        Serial1.print(",");
        Serial1.println(chunkLen);
        
        unsigned long waitPrompt = millis();
        bool prompt = false;
        while (millis() - waitPrompt < 1000) { 
            if (Serial1.find(">")) { prompt=true; break; } 
        }
        
        if(prompt) {
            Serial1.print(buf);
        } else {
            Serial.println("Send Error: Aborting!");
            return false;
        }
        delay(100); 
    }
    return true;
}

// Helper to extract IP from CIFSR response
String extractIP(String response) {
    // Example: +CIFSR:STAIP,"192.168.1.45"
    int staIpIdx = response.indexOf("STAIP,\"");
    if(staIpIdx != -1) {
        int start = staIpIdx + 7; // Skip STAIP,"
        int end = response.indexOf("\"", start);
        if(end != -1) return response.substring(start, end);
    }
    return "0.0.0.0";
}

    delay(2000);

    sendAT("AT+CIPSTO=2", 1000); 
}

void checkWebClients() {
    if (Serial1.available()) {
        String buffer = "";
        char c;
        unsigned long startTime = millis();
        while (millis() - startTime < 100) { 
            if (Serial1.available()) {
                c = Serial1.read();
                buffer += c;
            } else { delay(1); }
        }
        
        if (buffer.indexOf("+IPD,") != -1) {
            digitalWrite(LED_PIN, HIGH); 
            
            int startIp = buffer.indexOf("+IPD,") + 5;
            int endIp = buffer.indexOf(",", startIp);
            String idStr = buffer.substring(startIp, endIp);
            int connectionId = idStr.toInt();
            
            int getPos = buffer.indexOf("GET");
            String request = "";
            if (getPos != -1) request = buffer.substring(getPos);

            // --- DATA API ---
            if (request.indexOf("/api/data") != -1) {
                String json = "{";
                json += "\"temperature\":" + String(data_ref.temp, 1) + ",";
                json += "\"humidity\":" + String(data_ref.hum, 0) + ",";
                json += "\"airPressure\":" + String(data_ref.press, 1) + ",";
                json += "\"solarRadiation\":" + String(res_ghi, 0) + ",";
                json += "\"heatFlux\":" + String(res_flux, 1) + ",";
                json += "\"deltaTemperature\":" + String(res_delta_t, 2) + ",";
                // SEND THE NEW IP IN THE DATA PACKET
                json += "\"ip\":\"" + currentStationIP + "\"";
                json += "}";

                String header = "HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nConnection: close\r\n";
                header += "Content-Length: " + String(json.length()) + "\r\n\r\n";
                
                Serial1.print("AT+CIPSEND=");
                Serial1.print(connectionId);
                Serial1.print(",");
                Serial1.println(header.length());
                if(Serial1.find(">")) Serial1.print(header);
                delay(100);

                sendChunked(connectionId, json.c_str(), false);
                Serial1.print("AT+CIPCLOSE=");
                Serial1.println(connectionId);
                
            } 
            // --- NEW: WIFI CONNECT API (FIXED LOGIC) ---
            else if (request.indexOf("/api/wifi") != -1) {
                 String ssid = getUrlParam(request, "ssid");
                 String pass = getUrlParam(request, "pass");
                 
                 // 1. Send Response FIRST
                 String resp = "{\"status\":\"Connecting...\"}";
                 String header = "HTTP/1.1 200 OK\r\nContent-Type: application/json\r\nConnection: close\r\n";
                 header += "Content-Length: " + String(resp.length()) + "\r\n\r\n";
                 
                 Serial1.print("AT+CIPSEND=");
                 Serial1.print(connectionId);
                 Serial1.print(",");
                 Serial1.println(header.length());
                 if(Serial1.find(">")) Serial1.print(header);
                 delay(50);
                 sendChunked(connectionId, resp.c_str(), false);
                 Serial1.print("AT+CIPCLOSE=");
                 Serial1.println(connectionId);
                 
                 // 2. WAIT FOR CLOSE TO COMPLETE
                 delay(1000); 
                 while(Serial1.available()) Serial1.read(); // Flush junk

                 // 3. EXECUTE CONNECTION (BLOCKING)
                 if(ssid.length() > 0) {
                     Serial.print("\nAttempting connection to: "); Serial.println(ssid);
                     
                     // Enable Hybrid Mode
                     sendATWaitForResponse("AT+CWMODE=3", 2000);
                     delay(1000);
                     
                     // Send Connection Command and WAIT
                     String connectCmd = "AT+CWJAP=\"" + ssid + "\",\"" + pass + "\"";
                     
                     // Wait up to 15 seconds for connection result
                     if(sendATWaitForResponse(connectCmd, 15000)) {
                         Serial.println("\nSUCCESS: Wi-Fi Connected!");
                         
                         // --- GET NEW IP ADDRESS ---
                         Serial.println("GETTING NEW IP ADDRESS...");
                         delay(500);
                         Serial1.println("AT+CIFSR");
                         unsigned long ipStart = millis();
                         String ipResponse = "";
                         while(millis() - ipStart < 2000) { 
                            if(Serial1.available()) {
                                char c = Serial1.read();
                                ipResponse += c;
                                Serial.write(c);
                            }
                         }
                         // PARSE IP FROM RESPONSE
                         currentStationIP = extractIP(ipResponse);
                         Serial.print("NEW IP DETECTED: "); Serial.println(currentStationIP);
                         // --------------------------

                     } else {
                         Serial.println("\nFAILURE: Wi-Fi Connection Failed.");
                         currentStationIP = "0.0.0.0"; // Reset on fail
                     }
                     
                     // Flush any remaining data to prevent next loop reading junk
                     while(Serial1.available()) Serial1.read();
                 }
            }
            // --- WEBSITE RESPONSE ---
            else {
                int totalLen = strlen_P(HTML_PART_1) + strlen_P(LOGO_DATA_URI) + 
                               strlen_P(HTML_PART_2) + strlen_P(LOGO_DATA_URI) + 
                               strlen_P(HTML_PART_3) + strlen_P(LOGO_DATA_URI) +
                               strlen_P(HTML_PART_4);

                String header = "HTTP/1.1 200 OK\r\nContent-Type: text/html\r\nConnection: close\r\n";
                header += "Content-Length: " + String(totalLen) + "\r\n\r\n";
                
                Serial1.print("AT+CIPSEND=");
                Serial1.print(connectionId);
                Serial1.print(",");
                Serial1.println(header.length());
                
                if(Serial1.find(">")) Serial1.print(header);
                delay(100);

                if(sendChunked(connectionId, HTML_PART_1, true)) {
                    if(sendChunked(connectionId, LOGO_DATA_URI, true)) {
                        if(sendChunked(connectionId, HTML_PART_2, true)) {
                            if(sendChunked(connectionId, LOGO_DATA_URI, true)) {
                                if(sendChunked(connectionId, HTML_PART_3, true)) {
                                    if(sendChunked(connectionId, LOGO_DATA_URI, true)) {
                                        sendChunked(connectionId, HTML_PART_4, true);
                                    }
                                }
                            }
                        }
                    }
                }

                Serial1.print("AT+CIPCLOSE=");
                Serial1.println(connectionId);
            }
            digitalWrite(LED_PIN, LOW); 
        }
    }
}

void setup() {
    pinMode(LED_PIN, OUTPUT);
    digitalWrite(LED_PIN, HIGH); 
    
    Serial.begin(115200); 
    delay(2000);
    Serial.println("FiaPhy System Starting...");

    Wire.setSDA(PIN_SDA);
    Wire.setSCL(PIN_SCL);
    Wire.begin();

    if (!bme_ref.begin(0x76) || !bme_flux.begin(0x77)) {
        Serial.println("SENSOR ERROR!");
        while (1) { digitalWrite(LED_PIN, !digitalRead(LED_PIN)); delay(100); }
    }

    bme_ref.setSampling(Adafruit_BME280::MODE_FORCED, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::FILTER_OFF);
    bme_flux.setSampling(Adafruit_BME280::MODE_FORCED, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::FILTER_OFF);

    dtdss.configure(LATITUDE, LONGITUDE, ALTITUDE_M);
    
    bme_ref.takeForcedMeasurement();
    bme_flux.takeForcedMeasurement();
    delay(100);

    data_ref.temp = bme_ref.readTemperature();
    data_ref.hum = bme_ref.readHumidity();
    data_ref.press = bme_ref.readPressure() / 100.0f;
    
    softStartSystem(data_ref.temp, data_ref.hum, data_ref.press);
    
    initESP();
    digitalWrite(LED_PIN, LOW); 
    Serial.println("SYSTEM READY. CONNECT TO: " WIFI_SSID);
}

void loop() {
    checkWebClients();

    unsigned long current_time = millis();

    if (!waiting_for_sensor_read && (current_time - last_trigger_time >= SAMPLE_INTERVAL_MS)) {
        bme_ref.takeForcedMeasurement();
        bme_flux.takeForcedMeasurement();
        sensor_read_start_time = current_time;
        waiting_for_sensor_read = true;
    }

    if (waiting_for_sensor_read && (current_time - sensor_read_start_time >= 50)) {
        last_trigger_time = current_time; 
        waiting_for_sensor_read = false;

        float raw_ref_t = bme_ref.readTemperature();
        float raw_ref_h = bme_ref.readHumidity();
        float raw_ref_p = bme_ref.readPressure() / 100.0f;
        float raw_flux_t = bme_flux.readTemperature();
        float raw_flux_h = bme_flux.readHumidity();
        float raw_flux_p = bme_flux.readPressure() / 100.0f;

        data_ref.temp = apply_filter(raw_ref_t, data_ref.temp);
        data_ref.hum = apply_filter(raw_ref_h, data_ref.hum);
        data_ref.press = apply_filter(raw_ref_p, data_ref.press);
        data_flux.temp = apply_filter(raw_flux_t, data_flux.temp);
        data_flux.hum = apply_filter(raw_flux_h, data_flux.hum);
        data_flux.press = apply_filter(raw_flux_p, data_flux.press);

        dtdss.feedReferenceTemperature(data_ref.temp, 0);
        dtdss.feedReferenceHumidity(data_ref.hum, 0);
        dtdss.feedReferencePressure(data_ref.press, 0);
        dtdss.feedFluxTemperature(data_flux.temp, 0);
        dtdss.feedFluxHumidity(data_flux.hum, 0);
        dtdss.feedFluxPressure(data_flux.press, 0);

        if (dtdss.isFrameReady()) {
            FiaPhy::RadiationResult result = dtdss.compute();
            if (result.valid) {
                res_ghi = result.ghi_Wm2;          
                res_flux = result.heat_flux_Wm2;   
                res_delta_t = result.temp_differential_C;
                data_valid = true;
            }
        }
    }
